cmake_minimum_required(VERSION 3.24)
project(lmhtml VERSION 0.7.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMHTML_FULL_STATIC "Link the complete executable statically" ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(LMHTML_HOST_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmhtml.host.json")
set(LMHTML_STD_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/cpp/std/lmcode-std.schema.json")
set(LMHTML_MERGE_HELPER "${CMAKE_CURRENT_SOURCE_DIR}/scripts/merge-std.py")
set(LMHTML_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmhtml.dsl.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LMHTML_HOST_SCHEMA}" "${LMHTML_STD_SCHEMA}" "${LMHTML_MERGE_HELPER}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${LMHTML_MERGE_HELPER}" "${LMHTML_HOST_SCHEMA}" "${LMHTML_SCHEMA}" RESULT_VARIABLE LMHTML_MERGE_RESULT ERROR_VARIABLE LMHTML_MERGE_ERROR)
if(NOT LMHTML_MERGE_RESULT EQUAL 0)
  message(FATAL_ERROR "failed to merge lmcode std schema: ${LMHTML_MERGE_ERROR}")
endif()
file(READ "${LMHTML_SCHEMA}" LMHTML_SCHEMA_HEX HEX)
string(LENGTH "${LMHTML_SCHEMA_HEX}" LMHTML_SCHEMA_HEX_LENGTH)
math(EXPR LMHTML_SCHEMA_LAST_BYTE "${LMHTML_SCHEMA_HEX_LENGTH} - 2")
set(LMHTML_SCHEMA_INITIALIZER "")
foreach(offset RANGE 0 ${LMHTML_SCHEMA_LAST_BYTE} 2)
  string(SUBSTRING "${LMHTML_SCHEMA_HEX}" ${offset} 2 byte)
  string(APPEND LMHTML_SCHEMA_INITIALIZER "0x${byte},")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/schema_embedded.hpp"
  "#pragma once\n#include <cstddef>\nnamespace lmhtml { inline constexpr unsigned char embedded_schema[] = {${LMHTML_SCHEMA_INITIALIZER}}; inline constexpr std::size_t embedded_schema_size = sizeof(embedded_schema); }\n")
set(LEXBOR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEXBOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/lexbor)
add_executable(lmhtml cpp/main.cpp cpp/html.cpp cpp/strlib.cpp cpp/analyze.cpp cpp/plain.cpp cpp/markdown_input.cpp cpp/md4c/md4c.c cpp/md4c/md4c-html.c cpp/md4c/entity.c)
target_compile_definitions(lmhtml PRIVATE LMHTML_VERSION="${PROJECT_VERSION}")
target_include_directories(lmhtml PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}/generated" third_party/lexbor/source)
target_link_libraries(lmhtml PRIVATE lexbor_static)
target_compile_options(lmhtml PRIVATE -Wall -Wextra -Wpedantic)
# "Clang" also matches AppleClang; macOS has no -static/-static-libstdc++.
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmhtml PRIVATE -static-libstdc++ -static-libgcc)
  if(LMHTML_FULL_STATIC)
    target_link_options(lmhtml PRIVATE -static)
  endif()
endif()
